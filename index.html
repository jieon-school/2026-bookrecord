function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    var records = payload.records || [];

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["학년도", "학년", "반", "번호", "이름", "생기부 기재 규격 텍스트 (NEIS용)", "도서 수", "도서 상세 목록", "검증 상태", "제출일시", "교사 메모"]);
      sheet.getRange(1, 1, 1, 11).setBackground("#4F46E5").setFontColor("#FFFFFF").setFontWeight("bold");
    }

    if (action === "sync_all") {
      sheet.clearContents();
      sheet.appendRow(["학년도", "학년", "반", "번호", "이름", "생기부 기재 규격 텍스트 (NEIS용)", "도서 수", "도서 상세 목록", "검증 상태", "제출일시", "교사 메모"]);
      sheet.getRange(1, 1, 1, 11).setBackground("#4F46E5").setFontColor("#FFFFFF").setFontWeight("bold");

      records.forEach(function(r) {
        var neisText = (r.books || []).map(function(b) { return b.formattedString; }).join(", ");
        var detailText = (r.books || []).map(function(b, idx) { return (idx + 1) + ". " + b.formattedString; }).join("\n");
        sheet.appendRow([r.academicYear, r.studentGrade + "학년", r.studentClass + "반", r.studentNumber + "번", r.studentName, neisText, (r.books || []).length, detailText, r.status === "verified" ? "검증 완료" : "확인 필요", new Date(r.submittedAt).toLocaleString("ko-KR"), r.teacherComment || ""]);
      });
    } else if (action === "append_one") {
      var r = payload.record;
      var neisText = (r.books || []).map(function(b) { return b.formattedString; }).join(", ");
      var detailText = (r.books || []).map(function(b, idx) { return (idx + 1) + ". " + b.formattedString; }).join("\n");
      sheet.appendRow([r.academicYear, r.studentGrade + "학년", r.studentClass + "반", r.studentNumber + "번", r.studentName, neisText, (r.books || []).length, detailText, r.status === "verified" ? "검증 완료" : "확인 필요", new Date(r.submittedAt).toLocaleString("ko-KR"), r.teacherComment || ""]);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
